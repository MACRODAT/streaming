  // ── RENDER CARD CANVAS GEOMETRIES (TRIANGLES / POLYGONS) ──
  function drawCardCanvas(canvasId) {
    const canvas = document.getElementById(canvasId);
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    const colors = canvas.dataset.colors.split(',');
    
    // Set internal resolution match
    const w = canvas.width = canvas.offsetWidth * 2;
    const h = canvas.height = canvas.offsetHeight * 2;
    
    const totalItems = colors.length;
    
    ctx.clearRect(0, 0, w, h);
    
    if (totalItems === 4) {
      // Top Triangle
      ctx.fillStyle = colors[0]; ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(w,0); ctx.lineTo(w/2,h/2); ctx.fill();
      // Right Triangle
      ctx.fillStyle = colors[1]; ctx.beginPath(); ctx.moveTo(w,0); ctx.lineTo(w,h); ctx.lineTo(w/2,h/2); ctx.fill();
      // Bottom Triangle
      ctx.fillStyle = colors[2]; ctx.beginPath(); ctx.moveTo(w,h); ctx.lineTo(0,h); ctx.lineTo(w/2,h/2); ctx.fill();
      // Left Triangle
      ctx.fillStyle = colors[3]; ctx.beginPath(); ctx.moveTo(0,h); ctx.lineTo(0,0); ctx.lineTo(w/2,h/2); ctx.fill();
    } else if (totalItems === 3) {
      // 3 vertical diagonal cut panels
      ctx.fillStyle = colors[0]; ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(w*0.4,0); ctx.lineTo(w*0.2,h); ctx.lineTo(0,h); ctx.fill();
      ctx.fillStyle = colors[1]; ctx.beginPath(); ctx.moveTo(w*0.4,0); ctx.lineTo(w*0.8,0); ctx.lineTo(w*0.6,h); ctx.lineTo(w*0.2,h); ctx.fill();
      ctx.fillStyle = colors[2]; ctx.beginPath(); ctx.moveTo(w*0.8,0); ctx.lineTo(w,0); ctx.lineTo(w,h); ctx.lineTo(w*0.6,h); ctx.fill();
    } else if (totalItems === 2) {
      // Clean split angle cut rectangle
      ctx.fillStyle = colors[0]; ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(w*0.6,0); ctx.lineTo(w*0.4,h); ctx.lineTo(0,h); ctx.fill();
      ctx.fillStyle = colors[1]; ctx.beginPath(); ctx.moveTo(w*0.6,0); ctx.lineTo(w,0); ctx.lineTo(w,h); ctx.lineTo(w*0.4,h); ctx.fill();
    }
    
    // Overlay cinematic darkening mask over pure asset lines
    ctx.fillStyle = "rgba(0, 0, 0, 0.4)";
    ctx.fillRect(0, 0, w, h);
  }

  // ── CUSTOM INTERACTIVE ENGINE BUILDER ──
  let customDuration = 1;
  const MULTIPLIER_MATRIX = { 1: 0.80, 3: 0.75, 6: 0.70, 12: 0.60 };

  function toggleCustomSvc(element) {
    element.classList.toggle('selected');
    calculateCustomSummary();
  }

  function setCustomDuration(element) {
    document.querySelectorAll('.dur-chip').forEach(c => c.classList.remove('active'));
    element.classList.add('active');
    customDuration = parseInt(element.dataset.months);
    calculateCustomSummary();
  }

  function calculateCustomSummary() {
    const selectedElements = document.querySelectorAll('.check-btn.selected');
    const box = document.getElementById('custom-summary-box');
    
    if(selectedElements.length === 0) {
      box.innerHTML = "Select at least 1 service item panel.";
      return;
    }
    
    let listNames = [];
    let cumulativeBase = 0;
    
    selectedElements.forEach(el => {
      listNames.push(el.dataset.svc);
      cumulativeBase += parseFloat(el.dataset.price);
    });
    
    const factor = MULTIPLIER_MATRIX[customDuration];
    const totalEstimate = (cumulativeBase * customDuration * factor).toFixed(2);
    
    let spanLabel = customDuration === 12 ? "1 Year" : `${customDuration} Month(s)`;
    box.innerHTML = `${listNames.join(' + ')} (${spanLabel})<br><strong>Total Estimate: $${totalEstimate}</strong>`;
  }

  function sendCustomOrder() {
    const selectedElements = document.querySelectorAll('.check-btn.selected');
    if(selectedElements.length === 0) { return alert('Select services first.'); }
    
    let listNames = [];
    selectedElements.forEach(el => listNames.push(el.dataset.svc));
    
    const msg = encodeURIComponent(`Hi! Custom package design target configuration setup confirmation request:\nServices: ${listNames.join(' + ')}\nPeriod: ${customDuration} months.`);
    window.open(`https://wa.me/1234567890?text=${msg}`, '_blank');
  }

  // ── RETENTION TIMEOUT POPUP INTERSTITIAL TRIGGER ──
  let idleTimer = setTimeout(showRetentionPopup, 12000); // 12-second threshold window

  function showRetentionPopup() {
    document.getElementById('retentionPopup').classList.add('show');
  }
  function closeRetentionPopup() {
    document.getElementById('retentionPopup').classList.remove('show');
  }

  // Tracking clear indicators if user actively explores components to delay intrusion windows
  function resetIdlePattern() {
    clearTimeout(idleTimer);
    // Restart only if it hasn't popped up yet
    if(!document.getElementById('retentionPopup').classList.contains('show')) {
      idleTimer = setTimeout(showRetentionPopup, 20000);
    }
  }
  window.addEventListener('scroll', resetIdlePattern);
  window.addEventListener('click', resetIdlePattern);
